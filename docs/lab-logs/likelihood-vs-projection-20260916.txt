     "μακριά και από τα δύο" = dist_music > 2 ΚΑΙ dist_speech > 2
     ΧΩΡΙΣ Z_CLAMP εδώ (δεν ήταν μέρος της συνταγής του Δ, δηλωμένο

⚠ ΔΗΛΩΜΕΝΟΣ ΠΕΡΙΟΡΙΣΜΟΣ, ΠΡΙΝ ΤΟ ΤΡΕΞΙΜΟ: CV_POOLED_STD/FLUX_POOLED_STD
είναι ΜΙΑ κοινή (pooled) τυπική απόκλιση ανά άξονα — ΟΧΙ ξεχωριστή ανά
κλάση (music/speech). Το Δ ΔΕΝ είναι αληθινή Gaussian πιθανοφάνεια
(που θα χρειαζόταν σ_music και σ_speech ξεχωριστά ανά άξονα) — είναι
nearest-centroid με κοινή κλίμακα. Μαθηματική συνέπεια, ΕΠΑΛΗΘΕΥΜΕΝΗ
πριν το τρέξιμο: με κοινή κλίμακα και στις δύο κλάσεις, «πλησιέστερο
κέντρο» = ακριβώς η ΙΔΙΑ απόφαση με την προβολή στην ευθεία που
συνδέει τα κέντρα και σύγκριση με το μέσο (κλασική ταυτότητα: το
πρόσημο του (P-midpoint)·(B-A) καθορίζει ΚΑΙ τα δύο). DELTA_CV/
DELTA_FLUX/DELTA_SQ επαληθεύτηκαν να ΕΙΝΑΙ ακριβώς οι z-scored
διαφορές κέντρων (π.χ. (SPEECH_CV-MUSIC_CV)/CV_POOLED_STD = 1.6146 ≈
DELTA_CV) — άρα το Α ΕΙΝΑΙ ήδη μαθηματικά nearest-centroid. Το Δ
αναμένεται να δώσει ΤΗΝ ΙΔΙΑ ετυμηγορία με το Α σε κάθε σημείο — η
μόνη νέα πληροφορία είναι η ΤΡΙΤΗ κατάσταση («μακριά και από τα δύο»),
που η προβολή δεν μπορεί να εκφράσει καθόλου.

── ΠΡΟΒΛΕΨΗ, ΓΡΑΜΜΕΝΗ ΠΡΙΝ ΤΟ ΤΡΕΞΙΜΟ ──

Το Β νικάει το Α στα (3)(4)(6) και χάνει στο (2). Το Δ δείχνει ότι τα
δοκίμια ΜΕ bed έχουν πολλά παράθυρα μακριά και από τα δύο κέντρα.
⚠ Το άγνωστο, δηλωμένο: αν τα κέντρα έχουν μόνο μία κοινή τυπική
απόκλιση (pooled, όχι ανά κλάση), το Δ είναι περιορισμένο — και βάσει
της ταυτότητας πάνω, αναμένεται Δ≡Α σε κάθε ετυμηγορία.

import re, math, sys

MUSIC_CV=0.4375; SPEECH_CV=0.6855; CV_POOLED_STD=0.1536
MUSIC_FLUX=1.3387; SPEECH_FLUX=1.7242; FLUX_POOLED_STD=0.1738
DELTA_CV=1.61458; DELTA_FLUX=2.21807; DELTA_SQ=7.52673
Z_CLAMP=3.0

MID_FLUX=(MUSIC_FLUX+SPEECH_FLUX)/2.0
MID_CV=(MUSIC_CV+SPEECH_CV)/2.0

def clamp(x,lo,hi): return max(lo,min(hi,x))

def verdict_A(cv, flux):
    p_cv = clamp((cv-MUSIC_CV)/CV_POOLED_STD, -Z_CLAMP, Z_CLAMP)
    p_flux = clamp((flux-MUSIC_FLUX)/FLUX_POOLED_STD, -Z_CLAMP, Z_CLAMP)
    t = (p_cv*DELTA_CV + p_flux*DELTA_FLUX)/DELTA_SQ
    leaning = clamp(t,0.0,1.0)
    return "Speech" if leaning>=0.5 else "Music", t

def verdict_B(cv, flux):
    return "Speech" if flux>=MID_FLUX else "Music"

def verdict_G(cv, flux):
    return "Speech" if cv>=MID_CV else "Music"

def dist(cv,flux,c_cv,c_flux):
    return math.sqrt(((cv-c_cv)/CV_POOLED_STD)**2 + ((flux-c_flux)/FLUX_POOLED_STD)**2)

def verdict_D(cv, flux):
    dm = dist(cv,flux,MUSIC_CV,MUSIC_FLUX)
    ds = dist(cv,flux,SPEECH_CV,SPEECH_FLUX)
    v = "Speech" if ds<dm else "Music"
    certainty = abs(dm-ds)
    far_both = (dm>2.0 and ds>2.0)
    return v, dm, ds, certainty, far_both

cases = [
    ("1 ΣΚΕΤΗ ΑΦΗΓΗΣΗ", "test1", "AllSpeech"),
    ("2 ΣΚΕΤΟ BED", "test2", "AllMusic"),
    ("3 ΑΦΗΓΗΣΗ+BED -20dB", "test3", "AllSpeech"),
    ("4 ΑΦΗΓΗΣΗ+BED -12dB", "test4", "AllSpeech"),
    ("5 ΕΝΑΛΛΑΓΗ 30/60/90", "test5", "Alternating"),
    ("6 BED ΜΠΑΙΝΕΙ ΣΤΑ 60s", "test6", "AllSpeech"),
]

def expected(gt, center):
    if gt=="AllMusic": return "Music"
    if gt=="Alternating":
        block = int(center//30.0)
        return "Speech" if block%2==0 else "Music"
    return "Speech"

# parse raw output
text = open("/tmp/attack_detector_output.txt", encoding="utf-8").read()
sections = re.split(r"=== (\d) .*? \((test\d)_[^)]*\.wav\) ===", text)
# sections[0] is preamble; then triples (id, name, body)
data = {}
i = 1
while i < len(sections):
    idx = sections[i]; name = sections[i+1]; body = sections[i+2]
    rows = []
    for line in body.splitlines():
        m = re.match(r"\s*([\d.]+)s\s+επιθ=\s*(\d+)\s+ioi=\s*(\d+)\s+cv_ioi=\s*([\d.]+|NaN)\s+flux=\s*([\d.]+)\s+(ναι|ΟΧΙ)", line)
        if m:
            t, onsets, ioi, cv, flux, ok = m.groups()
            if cv == "NaN": continue
            rows.append((float(t), int(onsets), int(ioi), float(cv), float(flux), ok=="ναι"))
    data[name] = rows
    i += 3

print(f"MID_CV={MID_CV:.4f}  MID_FLUX={MID_FLUX:.4f}")
print()

overall = {"A":0,"B":0,"G":0,"D":0}
overall_n = 0
a_d_mismatches = 0
far_both_total = 0
far_both_by_test = {}

for label, key, gt in cases:
    rows = data[key]
    n = len(rows)
    correct = {"A":0,"B":0,"G":0,"D":0}
    far_both = 0
    for (t, onsets, ioi, cv, flux, logged_ok) in rows:
        center = t + 2.5
        exp = expected(gt, center)
        va, tval = verdict_A(cv, flux)
        vb = verdict_B(cv, flux)
        vg = verdict_G(cv, flux)
        vd, dm, ds, cert, fb = verdict_D(cv, flux)
        if va == exp: correct["A"] += 1
        if vb == exp: correct["B"] += 1
        if vg == exp: correct["G"] += 1
        if vd == exp: correct["D"] += 1
        if va != vd: a_d_mismatches += 1
        if fb: far_both += 1
        # sanity: va should match logged_ok
        logged_exp_ok = (va=="Speech") == logged_ok if exp=="Speech" else None
    far_both_by_test[label] = (far_both, n)
    far_both_total += far_both
    print(f"{label:<24} n={n:<4} Α={100*correct['A']/n:5.1f}%  Β={100*correct['B']/n:5.1f}%  Γ={100*correct['G']/n:5.1f}%  Δ={100*correct['D']/n:5.1f}%   μακριά-και-απο-τα-δύο={far_both}/{n}")
    for k in overall: overall[k]+=correct[k]
    overall_n += n
