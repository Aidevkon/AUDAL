//! MEASURE 2026-09-16: το δυναμικό κατώφλι (β) στα εννιά πραγματικά
//! librivox-hq αρχεία. Read-only, ΜΗΔΕΝ αλλαγή στην παραγωγή. ΜΗΔΕΝ
//! γνωστή απάντηση εδώ — ΔΕΝ μετριέται ορθότητα, μόνο δομικά μεγέθη
//! (τμήματα, resets, flagged, <5s), ΣΗΜΕΡΙΝΟ κατώφλι 0.20 έναντι
//! δυναμικού, ΙΔΙΟΣ κανόνας απόφασης (η προβολή, compute_scout_decision)
//! και στις δύο στήλες — ΜΟΝΟ η είσοδος cv_ioi αλλάζει.
//!
//! Decode: pass0_decode_to_dump (ΙΔΙΟ με escalation_analysis.rs, F-107)
//! — αυτούσιο production decode path. Ο dump διαβάζεται απευθείας ως
//! raw interleaved f32 LE στερεοφωνικό (ΙΔΙΟ format με DumpSeekProvider,
//! decode_provider.rs:297-303 — [L 4 bytes][R 4 bytes] ανά frame,
//! 48kHz), όχι μέσω hound/WAV. ΚΑΙ ΟΙ ΔΥΟ στήλες περνάνε από τον ΙΔΙΟ
//! αναγνώστη (ίδιο mono buffer, ίδιο window/hop σάρωμα) — συγκρίσιμες
//! μεταξύ τους. Η ασυμφωνία 79-έναντι-75 (scan_file έναντι
//! run_trunk_pass, F-112) δεν αφορά εδώ: και οι δύο στήλες μοιράζονται
//! το ΙΔΙΟ scan_file-συμβατό μονοπάτι, καμία σύγκριση με run_trunk_pass.
//!
//! ΣΗΜΕΡΙΝΟ: SpectralFluxDetector μέσω SegmentScout::measure (fixed
//! threshold 0.20, ΤΟ ΠΡΑΓΜΑΤΙΚΟ όργανο, αναλλοίωτο).
//! ΔΥΝΑΜΙΚΟ: αντίγραφο, γραμμή-προς-γραμμή, του SpectralFluxDetector::
//! detect (ΙΔΙΟ StftEngine, ΙΔΙΟ FLUX_MIN_DISTANCE) με κατώφλι
//!   threshold = median(flux_norm) + 2.0 · std(flux_norm)
//! τοπικά ανά παράθυρο 5s — ΙΔΙΟΣ τύπος με το F-112 (band_masked_
//! attack.rs), ΧΩΡΙΣ μάσκα bins (όλα τα N_BINS, ίδιο με «β» εκεί).
//! Το cepstral_flux ΜΕΝΕΙ αναλλοίωτο και στις δύο στήλες — ξεχωριστή
//! διαδρομή (MFCC ring), επαληθευμένο στον κώδικα.
//!
//! Κανόνας απόφασης: ο ΣΗΜΕΡΙΝΟΣ — compute_scout_decision πάνω σε
//! ScoutMeasurements{cv_ioi: <στήλη>, cepstral_flux: <ΙΔΙΟ και στις
//! δύο>}, μετά smooth_and_segment (ο τεμαχιστής, καμία αλλαγή). Το
//! flagged χρησιμοποιεί το avg_leaning/avg_confidence που ΗΔΗ
//! επιστρέφει το smooth_and_segment για κάθε στήλη ξεχωριστά (F-041
//! κατώφλια, αυτούσια).
//!
//! ΧΡΗΣΗ: cargo run --release --bin real_files_battery

use lineos_corpus::scout::{compute_cepstral_flux, compute_scout_decision, smooth_and_segment, ScoutDecision, SegmentBoundary, SegmentType, ScoutMeasurements};
use sp314_dsp::analysis::scout::SegmentScout;
use sp314_dsp::stft::spectral_flux::FLUX_MIN_DISTANCE;
use sp314_dsp::stft::{StftEngine, HOP_SIZE, N_BINS};

const WINDOW_SECS: f32 = 5.0;
const HOP_SECS: f32 = 1.0;
// ΑΥΤΟΥΣΙΑ scout.rs:235-237
const A7_ZONE_LOW: f32 = 0.3;
const A7_ZONE_HIGH: f32 = 0.7;
const A7_MIN_CONF: f32 = 0.4;
const DUMP_FRAME_BYTES: usize = 8; // 2 × f32 LE, decode_provider.rs
const DYNAMIC_K: f32 = 2.0;

fn median_std(xs: &[f32]) -> (f32, f32) {
    let mut sorted = xs.to_vec();
    sorted.sort_by(|a, b| a.partial_cmp(b).unwrap());
    let n = sorted.len();
    let med = if n == 0 {
        0.0
    } else if n % 2 == 0 {
        (sorted[n / 2 - 1] + sorted[n / 2]) / 2.0
    } else {
        sorted[n / 2]
    };
    let mean = xs.iter().sum::<f32>() / n.max(1) as f32;
    let var = xs.iter().map(|x| (x - mean).powi(2)).sum::<f32>() / n.max(1) as f32;
    (med, var.sqrt())
}

/// ΑΝΤΙΓΡΑΦΟ spectral_flux.rs:29-82, ΧΩΡΙΣ μάσκα bins (όλα τα
/// N_BINS, ίδιο με το «β» κελί του F-112) — ΜΟΝΗ αλλαγή: δυναμικό
/// κατώφλι αντί για fixed 0.20.
fn detect_dynamic(engine: &mut StftEngine, signal: &[f32]) -> Vec<usize> {
    let (frames, n_frames) = engine.forward(signal);
    let mut flux = vec![0.0_f32; n_frames];
    let mut prev = vec![0.0_f32; N_BINS];
    for (t, frame) in frames.iter().enumerate() {
        let mut frame_flux = 0.0_f32;
        for b in 0..N_BINS {
            let mag = (frame[b].re * frame[b].re + frame[b].im * frame[b].im).sqrt();
            let diff = mag - prev[b];
            if diff > 0.0_f32 {
                frame_flux += diff;
            }
            prev[b] = mag;
        }
        flux[t] = frame_flux;
    }
    let flux_max = flux.iter().cloned().fold(0.0_f32, f32::max);
    let mut flux_norm = vec![0.0_f32; n_frames];
    if flux_max > 1e-8_f32 {
        for t in 0..n_frames {
            flux_norm[t] = flux[t] / flux_max;
        }
    }
    let (med, std) = median_std(&flux_norm);
    let thr = med + DYNAMIC_K * std;

    let mut beats = Vec::new();
    for t in 1..n_frames.saturating_sub(1) {
        if flux_norm[t] >= thr && flux_norm[t] > flux_norm[t - 1] && flux_norm[t] > flux_norm[t + 1] {
            let dist = if beats.is_empty() { FLUX_MIN_DISTANCE + 1 } else { t - *beats.last().unwrap() };
            if dist >= FLUX_MIN_DISTANCE {
                beats.push(t);
            }
        }
    }
    beats
}

fn cv_ioi_from_onsets(onsets: &[usize], sample_rate: u32) -> f32 {
    if onsets.len() < 3 {
        return f32::NAN;
    }
    let hop_size = HOP_SIZE as f32;
    let mut iois = Vec::with_capacity(onsets.len() - 1);
    for i in 1..onsets.len() {
        let diff_frames = (onsets[i] - onsets[i - 1]) as f32;
        iois.push(diff_frames * hop_size * 1000.0 / sample_rate as f32);
    }
    let mean_ioi = iois.iter().sum::<f32>() / iois.len() as f32;
    if mean_ioi <= 1e-8 {
        return f32::NAN;
    }
    let mut sum_sq = 0.0;
    for &ioi in &iois {
        let diff = ioi - mean_ioi;
        sum_sq += diff * diff;
    }
    let variance = sum_sq / iois.len() as f32;
    libm::sqrtf(variance) / mean_ioi
}

fn read_dump_stereo(path: &str) -> (Vec<f32>, Vec<f32>) {
    let buf = std::fs::read(path).unwrap_or_else(|e| panic!("read dump {path}: {e}"));
    let full_frames = buf.len() / DUMP_FRAME_BYTES;
    let mut left = Vec::with_capacity(full_frames);
    let mut right = Vec::with_capacity(full_frames);
    for i in 0..full_frames {
        let base = i * DUMP_FRAME_BYTES;
        left.push(f32::from_le_bytes([buf[base], buf[base + 1], buf[base + 2], buf[base + 3]]));
        right.push(f32::from_le_bytes([buf[base + 4], buf[base + 5], buf[base + 6], buf[base + 7]]));
    }
    (left, right)
}

fn flagged(leaning: f32, conf: f32) -> bool {
    let in_dead_zone = leaning > A7_ZONE_LOW && leaning < A7_ZONE_HIGH;
    let low_confidence = conf < A7_MIN_CONF;
    in_dead_zone && low_confidence
}

fn music_to_speech_resets(segs: &[(f32, f32, SegmentType)]) -> usize {
    segs.windows(2).filter(|w| w[0].2 == SegmentType::Music && w[1].2 == SegmentType::Speech).count()
}

fn under_5s(segs: &[(f32, f32, SegmentType)]) -> usize {
    segs.iter().filter(|(s, e, _)| (e - s) < 5.0).count()
}

fn real_segs(boundaries: &[SegmentBoundary]) -> Vec<(f32, f32, SegmentType)> {
    boundaries.iter().map(|b| (b.start_sec, b.end_sec, b.segment_type)).collect()
}

fn flagged_count_real(boundaries: &[SegmentBoundary]) -> usize {
    boundaries.iter().filter(|b| flagged(b.avg_leaning, b.avg_confidence)).count()
}

fn run(label: &str, path: &str) {
    let dump = format!("/tmp/real_files_battery_{}.raw", label.split_whitespace().next().unwrap_or("x"));
    m0d::dsp::input_lufs::pass0_decode_to_dump(std::path::Path::new(path), &dump)
        .unwrap_or_else(|e| panic!("pass0 decode {path}: {e}"));
    let (left, right) = read_dump_stereo(&dump);
    let _ = std::fs::remove_file(&dump);

    let mono: Vec<f32> = left.iter().zip(right.iter()).map(|(&l, &r)| (l + r) * 0.5).collect();
    let sample_rate = lineos_types::analysis::ANALYSIS_SAMPLE_RATE;
    let win_samples = (WINDOW_SECS * sample_rate as f32) as usize;
    let hop_samples = (HOP_SECS * sample_rate as f32) as usize;

    let mut scout = SegmentScout::new(); // ΣΗΜΕΡΙΝΟ, fixed 0.20, αναλλοίωτο
    let mut dyn_engine = StftEngine::new(); // ΔΥΝΑΜΙΚΟ, ξεχωριστό instance
    let mut start = 0usize;

    let mut decisions_today: Vec<(f32, ScoutDecision)> = Vec::new();
    let mut decisions_dynamic: Vec<(f32, ScoutDecision)> = Vec::new();

    while start + win_samples <= mono.len() {
        let end = start + win_samples;
        let mono_slice = &mono[start..end];
        let start_sec = start as f32 / sample_rate as f32;

        let mut mfcc_analyzer = lineos_corpus::mfcc::MfccAnalyzer::new();
        let mut mfccs = Vec::new();
        let mut f = 0;
        while f + 1024 <= mono_slice.len() {
            mfccs.push(mfcc_analyzer.compute(&mono_slice[f..f + 1024]));
            f += 512;
        }
        let cepstral_flux = compute_cepstral_flux(&mfccs); // ΙΔΙΟ και στις δύο στήλες

        // ΣΗΜΕΡΙΝΟ: το πραγματικό SegmentScout::measure (fixed 0.20)
        let meas_today = scout.measure(mono_slice, cepstral_flux, sample_rate);
        let dec_today = compute_scout_decision(&meas_today);

        // ΔΥΝΑΜΙΚΟ: ίδιο cepstral_flux, νέο cv_ioi από το δυναμικό κατώφλι
        let onsets_dyn = detect_dynamic(&mut dyn_engine, mono_slice);
        let cv_dyn = cv_ioi_from_onsets(&onsets_dyn, sample_rate);
        let meas_dyn = ScoutMeasurements { cv_ioi: cv_dyn, cepstral_flux };
        let dec_dyn = compute_scout_decision(&meas_dyn);

        decisions_today.push((start_sec, dec_today));
        decisions_dynamic.push((start_sec, dec_dyn));

        start += hop_samples;
    }

    let boundaries_today = smooth_and_segment(&decisions_today);
    let boundaries_dynamic = smooth_and_segment(&decisions_dynamic);
    let segs_today = real_segs(&boundaries_today);
    let segs_dynamic = real_segs(&boundaries_dynamic);

    println!("=== {label} ===");
    println!(
        "  {:<10} {:>10} {:>10} {:>10} {:>10}",
        "στήλη", "τμήματα", "resets(M→S)", "flagged", "<5s"
    );
    println!(
        "  {:<10} {:>10} {:>10} {:>10} {:>10}",
        "ΣΗΜΕΡΙΝΟ",
        segs_today.len(),
        music_to_speech_resets(&segs_today),
        flagged_count_real(&boundaries_today),
        under_5s(&segs_today)
    );
    println!(
        "  {:<10} {:>10} {:>10} {:>10} {:>10}",
        "ΔΥΝΑΜΙΚΟ",
        segs_dynamic.len(),
        music_to_speech_resets(&segs_dynamic),
        flagged_count_real(&boundaries_dynamic),
        under_5s(&segs_dynamic)
    );
    println!();
}
